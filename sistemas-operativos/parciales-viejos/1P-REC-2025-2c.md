## Ejercicio 1

En una central atómica se desea controlar el estado del reactor nuclear. Hay un sistema de mediciones que escribe periódicamente los valores medidos por `stdout`. Para analizar y controlar dicha salida, se tienen dos funciones:
- `float costosa(float value);`
- `float aproximada(float value);`

a) Se desea implementar un programa que utilice dos subprocesos hijos para computar concurrentemente de ambas maneras (un hijo debe aplicar la función `costosa()` y el otro la función `aproximada()`, sobre cada medición). El padre a su vez debe estar constantemente obteniendo valores de `stdin`. Puede utilizar `scanf("%f", &x)` para leer un dato de `stdin` y convertirlo a un `float`. Cada hijo es responsable de escribir a `stdout` los resultados de sus propios cómputos.

b) Modificar el programa anterior utilizando las señales `SIGUSR1` y `SIGUSR2` de modo tal que cada subproceso notifique al proceso adre tan pronto como tenga un resultado, y sea el padre quien muestre todos los resultados por `stdout`.

### 1.A

```c
enum { READ, WRITE };
int pipe_aproximada[2];
int pipe_costosa[2];

void ejecutar_aproximado() {
	close(pipe_costosa[READ]);
	close(pipe_costosa[WRITE]);
	close(pipe_aproximada[WRITE]);
	float medicion_aprox;
	float x;
	while(read(pipe_aproximada[READ], &x, sizeof(x))) {
		medicion_aprox = aproximada(x);
		printf("Medición aproximada: %f\n", x);
	}
	exit(EXIT_SUCCESS);
}

void ejecutar_costosa() {
	close(pipe_aproximada[WRITE]);
	close(pipe_aproximada[READ]);
	close(pipe_costosa[WRITE]);
	
	float x;
	float medicion_costosa;
	while(read(pipe_costosa[READ], &x, sizeof(x))) {
		medicion_costosa = costosa(x);
		printf("Medición costosa: %f\n", x);
	}
	exit(EXIT_SUCCESS);
}

int main() {
	pipe(pipe_aproximada);
	pipe(pipe_costosa);
	
	if (fork() == 0) {
		ejecutar_aproximado();
		exit(EXIT_FAILURE); // si salió sin ejecutar la función.
	}
	
	if (fork() == 0) {
		ejecutar_costosa();
		exit(EXIT_FAILURE);
	}
	
	close(pipe_aproximada[READ]);
	close(pipe_costosa[READ]);
	float x;
	
	for (;;) {
		scanf("%f", &x);
		write(pipe_aproximada[WRITE], &x, sizeof(x));
		write(pipe_costosa[WRITE], &x, sizeof(x));
	}
	
	return EXIT_SUCCESS;
}
```

### 1.B

```c
enum { READ, WRITE };

int pipe_aproximada_ph[2];
int pipe_costosa_ph[2];
int pipe_aproximada_hp[2];
int pipe_costosa_hp[2];

volatile sigatomic_t costosa_lista = FALSE;
volatile sigatomic_t aproximada_lista = FALSE;

void avisar_padre_costosa(int sig) {
	costosa_lista = TRUE;
}

void avisar_padre_aproximada(int sig) {
	aproximada_lista = TRUE;
}

void ejecutar_aproximado() {
	close(pipe_costosa_hp[READ]);
	close(pipe_costosa_hp[WRITE]);
	close(pipe_costosa_ph[READ]);
	close(pipe_costosa_ph[WRITE]);
	
	close(pipe_aproximada_hp[READ]);
	close(pipe_aproximada_ph[WRITE]);
	
	float medicion_aprox;
	float x;
	while(read(pipe_aproximada_ph[READ], &x, sizeof(x)) > 0) {
		medicion_aprox = aproximada(x);
		write(pipe_aproximada_hp[WRITE], &medicion_aprox, sizeof(medicion_aprox));
		kill(getppid(), SIGUSR2);
	}
	exit(EXIT_SUCCESS);
}

void ejecutar_costosa() {
	close(pipe_aproximada_ph[WRITE]);
	close(pipe_aproximada_ph[READ]);
	close(pipe_aproximada_hp[WRITE]);
	close(pipe_aproximada_hp[READ]);
	
	close(pipe_costosa_hp[READ]);
	close(pipe_costosa_ph[WRITE]);
	
	float x;
	float medicion_costosa;
	while(read(pipe_costosa_ph[READ], &x, sizeof(x)) > 0) {
		medicion_costosa = costosa(x);
		write(pipe_costosa_hp[WRITE], &medicion_costosa, sizeof(medicion_costosa));
		kill(getppid(), SIGUSR1);
	}
	exit(EXIT_SUCCESS);
}

int main() {
	pipe(pipe_aproximada_ph);
	pipe(pipe_costosa_ph);
	pipe(pipe_aproximada_hp);
	pipe(pipe_costosa_hp);
	
	if (fork() == 0) {
		ejecutar_aproximado();
		exit(EXIT_FAILURE); // si salió sin ejecutar la función.
	}
	
	if (fork() == 0) {
		ejecutar_costosa();
		exit(EXIT_FAILURE);
	}
	
	close(pipe_aproximada_ph[READ]); close(pipe_aproximada_hp[WRITE]);
	close(pipe_costosa_ph[READ]); close(pipe_costosa_hp[WRITE]);
	
	signal(SIGUSR1, avisar_padre_costosa);
	signal(SIGUSR2, avisar_padre_aproximada);
	
	float x;
	float val;
	
	while (scanf("%f", &x) == 1) {
		aproximada_lista = FALSE;
		costosa_lista = FALSE;
		
		write(pipe_aproximada_ph[WRITE], &x, sizeof(x));
		write(pipe_costosa_ph[WRITE], &x, sizeof(x));
		
		int resultados_obtenidos = 0;
		
		while (resultados_obtenidos < 2) {
			pause();
			if (aproximada_lista) {
				read(pipe_aproximada_hp[READ], &val, sizeof(val));
				printf("Medición aproximada: %f", val);
				aproximada_lista = FALSE;
				resultados_obtenidos++;
			}
			if (costosa_lista) {
				read(pipe_costosa_hp[READ], &val, sizeof(val));
				printf("Medición costosa: %f", val);
				costosa_lista = FALSE;
				resultados_obtenidos++;
			}
		}
	}
	
	return EXIT_SUCCESS;
}
```
